  asr_decoder.cc
  asr_model.cc

  ctc_prefix_beam_search.cc
  ctc_wfst_beam_search.cc
  ctc_endpoint.cc
)

  message(FATAL_ERROR "Please build with TORCH or ONNX or TFLITE!!!")

  list(APPEND decoder_srcs torch_asr_model.cc)

  list(APPEND decoder_srcs onnx_asr_model.cc)
